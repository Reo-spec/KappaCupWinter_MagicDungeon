03Strategy ― 変わる部分だけを外に出す

// BulletMotion.h
/// <summary>弾の飛び方。</summary>
class IBulletMotion {
public:
    virtual ~IBulletMotion() {}

    /// <summary>撃たれたとき、1回だけ。</summary>
    virtual void OnFire(const Vector3& startPos, const Vector3& dir) {}

    /// <summary>毎フレーム、位置を動かす。</summary>
    virtual void Move(Vector3& position, float dt) = 0;
};

class StraightMotion : public IBulletMotion {
public:
    explicit StraightMotion(float speed) : m_speed(speed) {}
    void OnFire(const Vector3& startPos, const Vector3& dir) override
    {
        m_velocity = dir * m_speed;
    }
    void Move(Vector3& position, float dt) override
    {
        position += m_velocity * dt;
    }
private:
    float   m_speed;
    Vector3 m_velocity = Vector3::Zero;
};

class HomingMotion : public IBulletMotion {
public:
    HomingMotion(float speed, float turnRate)
        : m_speed(speed), m_turnRate(turnRate) {}

    void OnFire(const Vector3& startPos, const Vector3& dir) override
    {
        m_velocity = dir * m_speed;
    }
    void Move(Vector3& position, float dt) override
    {
        Enemy* target = FindNearestEnemy(position);
        if (target != nullptr) {
            Vector3 toTarget = target->GetPosition() - position;
            toTarget.Normalize();
            // 少しずつ、目標の方向へ寄せる
            m_velocity += toTarget * m_turnRate * dt;
            m_velocity.Normalize();
            m_velocity *= m_speed;
        }
        position += m_velocity * dt;
    }
private:
    float   m_speed, m_turnRate;
    Vector3 m_velocity = Vector3::Zero;
};
// Bullet.cpp ― 1つだけになる
void Bullet::Reset(const Vector3& pos, const Vector3& dir,
                   std::unique_ptr<IBulletMotion> motion)
{
    m_position = pos;
    m_motion   = std::move(motion);
    m_motion->OnFire(pos, dir);
    m_lifeTime = 0.0f;
}

void Bullet::Update()
{
    const float dt = g_gameTime->GetFrameDeltaTime();

    m_motion->Move(m_position, dt);     // ★ ここだけが差し替わる

    // 以下は全部の弾で共通
    m_lifeTime += dt;
    if (m_lifeTime > MAX_LIFE || IsOutOfScreen()) {
        Deactivate();
        return;
    }
    CheckHit();
    m_spriteRender.SetPosition(m_position);
    m_spriteRender.Update();
}

04State との違い

StrategyState
形どちらも「中身を差し替える」。形はほぼ同じ
いつ差し替わる 外から決められて、基本そのまま 自分で次の状態を決めて、自動で変わる
互いを知っているか 知らない 次の状態を知っている
例弾の軌道、敵の動き方、AIの難易度 ボスのフェーズ、キャラの待機／歩き／攻撃
迷ったら、どちらでも構いません コードの形はほとんど同じです。 名前の違いは「意図の違い」を表しているだけで、 間違えても動かなくなることはありません。
面接で聞かれたときに答えられれば充分です。

06毎回 new するのが気になる場合

std::make_unique を撃つたびに呼ぶと、 4-06 でやった問題が戻ってきます。

状態を持たない Strategy なら、使い回せます。 

// 速度だけ渡して、内部に状態を持たない形にする
class IBulletMotion {
public:
    virtual ~IBulletMotion() {}
    virtual void Move(Vector3& position, Vector3& velocity, float dt) const = 0;
};

class StraightMotion : public IBulletMotion {
public:
    void Move(Vector3& position, Vector3& velocity, float dt) const override
    {
        position += velocity * dt;
    }
};
// 1個ずつ作っておいて、ポインタだけ渡す
namespace {
    StraightMotion s_straight;
    HomingMotion   s_homing;
    WaveMotion     s_wave;
}

pool->Fire(pos, dir * 800.0f, &s_straight);
状態を持たせられるかどうかで、書き方が変わります 持たせるなら、弾ごとに1つ必要（unique_ptr）。
持たせないなら、1個を全員で共有できる（ポインタ）。
状態は弾側（velocity など）に置けないか、先に考えてください。

/// <summary>進行方向に対して、左右に波打ちながら飛ぶ。</summary>
class WaveMotion : public IBulletMotion {
public:
    WaveMotion(float speed, float width, float cycle)
        : m_speed(speed), m_width(width), m_cycle(cycle) {}

    void OnFire(const Vector3& startPos, const Vector3& dir) override
    {
        m_forward = dir;
        m_forward.Normalize();

        // 進行方向に対して垂直な向きを作る（Y軸まわりに90度回す）
        m_side = { -m_forward.z, 0.0f, m_forward.x };

        m_basePos = startPos;
        m_time    = 0.0f;
    }

    void Move(Vector3& position, float dt) override
    {
        m_time += dt;
        m_basePos += m_forward * m_speed * dt;

        float offset = sinf(m_time * m_cycle) * m_width;
        position = m_basePos + m_side * offset;
    }

private:
    float   m_speed, m_width, m_cycle;
    Vector3 m_forward = Vector3::Zero;
    Vector3 m_side    = Vector3::Zero;
    Vector3 m_basePos = Vector3::Zero;
    float   m_time    = 0.0f;
};

/// <summary>だんだん速くなる。</summary>
class AccelMotion : public IBulletMotion {
public:
    AccelMotion(float startSpeed, float accel, float maxSpeed)
        : m_speed(startSpeed), m_accel(accel), m_maxSpeed(maxSpeed) {}

    void OnFire(const Vector3& startPos, const Vector3& dir) override
    {
        m_dir = dir;
        m_dir.Normalize();
    }

    void Move(Vector3& position, float dt) override
    {
        m_speed += m_accel * dt;
        if (m_speed > m_maxSpeed) { m_speed = m_maxSpeed; }
        position += m_dir * m_speed * dt;
    }

private:
    float   m_speed, m_accel, m_maxSpeed;
    Vector3 m_dir = Vector3::Zero;
};

使う

pool->Fire(pos, dir, std::make_unique<WaveMotion>(400.0f, 80.0f, 8.0f));
pool->Fire(pos, dir, std::make_unique<AccelMotion>(100.0f, 900.0f, 1200.0f));

「まっすぐ進む中心（m_basePos）」を別に持って、 そこからのズレとして位置を決めるのが正解です。 波・回転・振動の実装で、ほぼ必ず出てくる落とし穴です。

おすすめは「敵の撃ち方」です

class IShotPattern {
public:
    virtual ~IShotPattern() {}
    /// <summary>1回撃つ。</summary>
    virtual void Shot(const Vector3& from, BulletPool& pool) = 0;
};

/// <summary>プレイヤーを狙って1発。</summary>
class AimShot : public IShotPattern {
public:
    void Shot(const Vector3& from, BulletPool& pool) override
    {
        Player* p = FindGO<Player>("player");
        if (p == nullptr) { return; }

        Vector3 dir = p->GetPosition() - from;
        dir.Normalize();
        pool.Fire(from, dir);
    }
};

/// <summary>扇状に n 発。</summary>
class FanShot : public IShotPattern {
public:
    FanShot(int num, float totalAngleDeg)
        : m_num(num), m_total(totalAngleDeg) {}

    void Shot(const Vector3& from, BulletPool& pool) override
    {
        if (m_num <= 0) { return; }

        float step  = (m_num == 1) ? 0.0f : m_total / (m_num - 1);
        float start = -m_total * 0.5f;

        for (int i = 0; i < m_num; i++) {
            Quaternion rot;
            rot.SetRotationDegY(start + step * i);

            Vector3 dir = { 0.0f, 0.0f, -1.0f };
            rot.Apply(dir);
            pool.Fire(from, dir);
        }
    }
private:
    int   m_num;
    float m_total;
};

/// <summary>全方位。</summary>
class CircleShot : public IShotPattern {
public:
    explicit CircleShot(int num) : m_num(num) {}
    void Shot(const Vector3& from, BulletPool& pool) override
    {
        for (int i = 0; i < m_num; i++) {
            Quaternion rot;
            rot.SetRotationDegY(360.0f / m_num * i);

            Vector3 dir = { 0.0f, 0.0f, -1.0f };
            rot.Apply(dir);
            pool.Fire(from, dir);
        }
    }
private:
    int m_num;
};

State × Strategy の組み合わせが効きます State が「いつ」を決め、Strategy が「何を」決める。

この2つを組み合わせると、 弾幕のパターンを増やすのが、クラスを1つ足すだけの作業になります。 ボス戦の作り込みが、一気に楽しくなるところです。

08確認

  1. Strategy と State の違いは？
  2. Strategy を使い回せるようにするには、何を避ける？
  3. 波打つ動きを書くとき、位置を直接揺らしてはいけないのはなぜ？
答え
  1. Strategy は外から決められてそのまま。 State は自分で次の状態を決めて自動で変わる。
  2. 内部に状態を持たせないこと。 持たなければ1個を全員で共有できる。
  3. 揺れが累積して発散するから。 中心の位置を別に持ち、そこからのズレとして決める。
