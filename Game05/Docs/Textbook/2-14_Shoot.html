02Bullet クラス

// Bullet.h
#pragma once

class Bullet : public IGameObject
{
public:
    bool Start() override;
    void Update() override;
    void Render(RenderContext& rc) override;

    void SetPosition(const Vector3& pos) { m_position = pos; }
    const Vector3& GetPosition() const   { return m_position; }
    float GetRadius() const              { return RADIUS; }

private:
    SpriteRender m_spriteRender;
    Vector3      m_position = Vector3::Zero;

    static const float SPEED;
    static const float RADIUS;
};
// Bullet.cpp
#include "stdafx.h"
#include "Bullet.h"

const float Bullet::SPEED  = 1200.0f;
const float Bullet::RADIUS = 8.0f;

bool Bullet::Start()
{
    m_spriteRender.Init("Assets/sprite/bullet.dds", 16.0f, 32.0f);
    return true;
}

void Bullet::Update()
{
    const float dt = g_gameTime->GetFrameDeltaTime();

    m_position.y += SPEED * dt;      // まっすぐ上へ

    // 画面の外に出たら消える。
    if (m_position.y > 600.0f) {
        DeleteGO(this);
        return;                      // ★ 忘れない
    }

    m_spriteRender.SetPosition(m_position);
    m_spriteRender.Update();
}

void Bullet::Render(RenderContext& rc)
{
    m_spriteRender.Draw(rc);
}
画面外で消す、を絶対に忘れない これを書かないと、弾は永遠に上に飛び続けて Update され続けます。 GameObject List の Alive が増え続けたら、それが起きています （2-06）。

03発射する

自機から NewGO するだけです。 

void Player::Shoot()
{
    Bullet* bullet = NewGO<Bullet>(enPriority_Bullet, "bullet");

    // 自機の少し上から出す。
    Vector3 pos = m_position;
    pos.y += 40.0f;
    bullet->SetPosition(pos);
}

ここで受け取った bullet ポインタは、その場で使うだけです。 メンバ変数に覚えておいてはいけません （覚えるなら GOHandle。2-09）。

04連射を制御する

ボタンに IsPress をそのままつなぐと、 1秒間に60発出ます。多すぎます。 クールタイムで間引きます。 

// Player.h
private:
    float m_shootCoolTime = 0.0f;
    static const float SHOOT_INTERVAL;   // 何秒ごとに撃てるか
// Player.cpp
const float Player::SHOOT_INTERVAL = 0.12f;   // 1秒に約8発

void Player::Shoot()
{
    const float dt = g_gameTime->GetFrameDeltaTime();

    // 時間を減らしていく。
    m_shootCoolTime -= dt;

    if (m_shootCoolTime > 0.0f) {
        return;                       // まだ撃てない
    }
    if (g_pad[0]->IsPress(enButtonA) == false) {
        return;                       // 押されていない
    }

    // 撃つ。
    Bullet* bullet = NewGO<Bullet>(enPriority_Bullet, "bullet");
    Vector3 pos = m_position;
    pos.y += 40.0f;
    bullet->SetPosition(pos);

    m_shootCoolTime = SHOOT_INTERVAL;   // 時間をリセット
}
「フレーム数を数える」のではなく「秒で数える」 if (m_frame % 8 == 0) のようにフレーム数で間引くと、 PCが重いときに連射速度が変わってしまいます。 2-02 と同じ理由です。秒で数えてください。

05動かして確かめる

  • Aボタンを押しっぱなしにすると、一定の間隔で弾が出る
  • SHOOT_INTERVAL を変えると連射速度が変わる
  • GameObject List の Alive が、一定のところで止まる

最後のひとつが大事です。ずっと増え続けていたら、画面外での DeleteGO が 効いていません。

06よくあるエラー

弾が1発も出ない

原因1：Shoot() を Update() から呼んでいない。
原因2：SetPosition のあとに m_spriteRender.Update() を呼んでいない。
原因3：弾が自機の後ろに隠れている。優先度を確認する。

弾が1発しか出ない

原因：m_shootCoolTime をリセットしていない、 または減らしていない。
直し方：減らす処理は return より前に書く。 後ろに書くと、一度撃てなくなったら永久に減りません。

だんだん重くなる

原因：画面外の DeleteGO 忘れ。
直し方：GameObject List で Alive を確認する。

演習2（増やす）

解答例を見る（まず自分でやってから）

弾に「進む向き」を持たせます。いままで y だけ足していたのを、 ベクトルで持つように変えます。 

// Bullet.h
public:
    void SetVelocity(const Vector3& v) { m_velocity = v; }
private:
    Vector3 m_velocity = { 0.0f, 1.0f, 0.0f };   // 向き（長さ1）

// Bullet.cpp
void Bullet::Update()
{
    const float dt = g_gameTime->GetFrameDeltaTime();
    m_position += m_velocity * SPEED * dt;
    ...
}
// Player.cpp
void Player::ShootWay3()
{
    const float ANGLE = 15.0f;    // 何度ずらすか

    for (int i = -1; i <= 1; i++) {
        Bullet* bullet = NewGO<Bullet>(enPriority_Bullet, "bullet");

        Vector3 pos = m_position;
        pos.y += 40.0f;
        bullet->SetPosition(pos);

        // 上向き(0,1,0)を、Z軸まわりに回す。
        Quaternion rot;
        rot.SetRotationDegZ(ANGLE * i);
        Vector3 dir = { 0.0f, 1.0f, 0.0f };
        rot.Apply(dir);
        bullet->SetVelocity(dir);
    }
}

回転を使わない書き方もあります。中学の三角関数です。

float rad = Math::DegToRad(90.0f + ANGLE * i);   // 90度＝真上
Vector3 dir = { cosf(rad), sinf(rad), 0.0f };
bullet->SetVelocity(dir);

どちらでも構いません。 この書き方が分かると、全方位に16発ばらまくのも ループの回数を変えるだけで作れます。ボスの攻撃に使えます。

演習3（自分で書く）

ボタンを押し続けるとだんだん連射が速くなるようにする。 ただし上限をつけること。 本文には答えがありません。

解答例を見る（まず自分でやってから） 
// Player.h
private:
    float m_shootCoolTime = 0.0f;
    float m_holdTime = 0.0f;         // 押し続けている時間
    static const float INTERVAL_SLOW;   // 最初の間隔
    static const float INTERVAL_FAST;   // 最速の間隔
    static const float SPEED_UP_TIME;   // 何秒で最速になるか

// Player.cpp
const float Player::INTERVAL_SLOW  = 0.20f;
const float Player::INTERVAL_FAST  = 0.06f;
const float Player::SPEED_UP_TIME  = 2.0f;

void Player::Shoot()
{
    const float dt = g_gameTime->GetFrameDeltaTime();
    m_shootCoolTime -= dt;

    if (g_pad[0]->IsPress(enButtonA) == false) {
        m_holdTime = 0.0f;           // 離したらリセット
        return;
    }
    m_holdTime += dt;

    if (m_shootCoolTime > 0.0f) {
        return;
    }

    // 押している時間に応じて 0.0 → 1.0 に近づく（1.0で止める）
    float rate = min(m_holdTime / SPEED_UP_TIME, 1.0f);
    // 間隔を SLOW から FAST へ、なめらかに近づける
    float interval = INTERVAL_SLOW + (INTERVAL_FAST - INTERVAL_SLOW) * rate;

    FireOne();
    m_shootCoolTime = interval;
}

ポイント

  • a + (b - a) * rate は線形補間という、 ゲームで非常によく使う式です。rate が 0 なら a、 1 なら b、0.5 ならその中間になります
  • min(..., 1.0f) で上限をつける。これを忘れると、 押し続けるほど間隔が0を通り越してマイナスになります

08確認

  1. クールタイムをフレーム数ではなく秒で数えるのはなぜ？
  2. 画面外の弾を消さないとどうなる？ どこで確認できる？
  3. NewGO が返したポインタを、メンバ変数に持ってよい？
答え
  1. PCの速さで連射速度が変わってしまうから。
  2. ずっと Update され続けて重くなる。GameObject List の Alive で分かる。
  3. 持ってはいけない。持つなら GOHandle。
